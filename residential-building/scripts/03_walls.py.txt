import clr
clr.AddReference('RevitServices')
from RevitServices.Persistence import DocumentManager
from RevitServices.Transactions import TransactionManager

clr.AddReference('RevitAPI')
from Autodesk.Revit.DB import (
    Wall, WallType, Level, Line, XYZ,
    FilteredElementCollector, BuiltInParameter
)

doc = DocumentManager.Instance.CurrentDBDocument
ft_per_m = 3.28084

levels = FilteredElementCollector(doc).OfClass(Level).ToElements()
ground_level = None
level_1 = None
for lvl in levels:
    if lvl.Name == "PRM-Ground":
        ground_level = lvl
    if lvl.Name == "PRM-Level 1":
        level_1 = lvl

wall_types = FilteredElementCollector(doc).OfClass(WallType).ToElements()
basic_wall_type = None
for wt in wall_types:
    if wt.Kind == wt.Kind.Basic:
        basic_wall_type = wt
        break
if basic_wall_type is None:
    basic_wall_type = wall_types[0]

width = 15.0
depth = 12.0

corners = [
    XYZ(0.0, 0.0, 0.0),
    XYZ(width * ft_per_m, 0.0, 0.0),
    XYZ(width * ft_per_m, depth * ft_per_m, 0.0),
    XYZ(0.0, depth * ft_per_m, 0.0)
]

TransactionManager.Instance.EnsureInTransaction(doc)

new_walls = []
for i in range(4):
    start = corners[i]
    end = corners[(i + 1) % 4]
    line = Line.CreateBound(start, end)
    wall = Wall.Create(doc, line, basic_wall_type.Id, ground_level.Id, 3.3 * ft_per_m, 0.0, False, False)
    wall.get_Parameter(BuiltInParameter.WALL_HEIGHT_TYPE).Set(level_1.Id)
    wall.Flip()
    new_walls.append(wall)

TransactionManager.Instance.TransactionTaskDone()

OUT = "Created {} perimeter walls, top constrained to {}, flipped for correct exterior/interior".format(len(new_walls), level_1.Name)
